### 阶段 2：理解服务器启动 (30 分钟)

#### 2.1 阅读 Server::start()

```cpp
// 文件位置：Server.cpp
// 函数：Server::start()
// 阅读步骤：

// 步骤 1：创建控制套接字
controlServerSocket = socket(AF_INET, SOCK_STREAM, 0);

// 步骤 2：设置套接字选项
setsockopt(controlServerSocket, SOL_SOCKET, SO_REUSEADDR, ...);

// 步骤 3：绑定控制端口
bind(controlServerSocket, (struct sockaddr*)&controlAddr, ...);

// 步骤 4：开始监听
listen(controlServerSocket, SOMAXCONN);

// 步骤 5：创建数据套接字（主动模式用）
dataServerSocket = socket(AF_INET, SOCK_STREAM, 0);
bind(dataServerSocket, ...);

// 步骤 6：创建 epoll
epollFd = epoll_create1(0);

// 步骤 7：添加控制套接字到 epoll
epoll_ctl(epollFd, EPOLL_CTL_ADD, controlServerSocket, &ev);
```

**关键点**：
- 控制端口 2100 用于命令传输
- 数据端口 2000 用于主动模式（本实现主要用被动模式）
- epoll 用于高效监听多个连接

#### 2.2 阅读 Server::run()

```cpp
// 文件位置：Server.cpp
// 函数：Server::run()
// 核心逻辑：

void Server::run() {
    while(running) {
        // 等待事件，超时 1 秒
        int nfds = epoll_wait(epollFd, events, MAX_EPOLL_EVENTS, 1000);
        
        for(int i = 0; i < nfds; i++) {
            if(events[i].data.fd == controlServerSocket) {
                // 新客户端连接
                int clientSock = accept(...);
                // 创建会话
                FtpSession* session = new FtpSession();
                // 添加到 epoll
                epoll_ctl(epollFd, EPOLL_CTL_ADD, clientSock, &ev);
            } else {
                // 客户端数据
                threadPool->submit([this, clientSock]() {
                    handleClientData(clientSock);
                });
            }
        }
    }
}
```

**关键点**：
- epoll_wait 阻塞等待事件
- 新连接：创建会话，添加到 epoll
- 数据：提交到线程池处理

### 阶段 3：理解命令处理 (45 分钟)

#### 3.1 阅读 handleClientData()

```cpp
// 文件位置：Server.cpp
// 函数：Server::handleClientData()

void Server::handleClientData(int clientSock) {
    char buffer[SIZE_BUFFER];
    
    // 接收数据
    size_t bytesReceived = recv(clientSock, buffer, sizeof(buffer) - 1, 0);
    
    // 检查连接是否关闭
    if(bytesReceived <= 0) {
        removeSession(clientSock);
        return;
    }
    
    // 解析命令
    buffer[bytesReceived] = '\0';
    string command = buffer;
    
    // 移除\r\n
    while(!command.empty() && (command.back() == '\r' || command.back() == '\n')) {
        command.pop_back();
    }
    
    // 处理命令
    handleFtpCommand(clientSock, command);
}
```

#### 3.2 阅读 handleFtpCommand()

```cpp
// 文件位置：Server.cpp
// 函数：Server::handleFtpCommand()
// 阅读重点：命令分发逻辑

void Server::handleFtpCommand(int clientSock, const string& command) {
    // 分割命令和参数
    string cmd, args;
    size_t space = command.find(' ');
    if(space != string::npos) {
        cmd = command.substr(0, space);
        args = command.substr(space + 1);
    } else {
        cmd = command;
    }
    
    // 转换为大写
    std::transform(cmd.begin(), cmd.end(), cmd.begin(), ::toupper);
    
    // 命令分发
    if(cmd == "PASV") {
        handlePasvCommand(clientSock);
    } else if(cmd == "LIST") {
        handleListCommand(clientSock);
    } else if(cmd == "RETR") {
        handleRetrCommand(clientSock, args);
    }
    // ... 其他命令
}
```

2. **handleListCommand()** - 列目录
3. **handleRetrCommand()** - 下载文件
4. **handleCwdCommand()** - 切换目录

```cpp
// 文件位置：Client.cpp
// 函数：Client::pasvMode()

bool Client::pasvMode() {
    // 发送 PASV 命令
    string response = sendFtpCommand("PASV");
    
    // 解析响应：227 (h1,h2,h3,h4,p1,p2)
    size_t start = response.find('(');
    size_t end = response.find(')');
    string params = response.substr(start + 1, end - start - 1);
    
    // 解析端口：port = p1*256 + p2
    int h1, h2, h3, h4, p1, p2;
    sscanf(params.c_str(), "%d,%d,%d,%d,%d,%d", &h1, &h2, &h3, &h4, &p1, &p2);
    
    pasvPort = p1 * 256 + p2;
    pasvHost = to_string(h1) + "." + to_string(h2) + "." + 
               to_string(h3) + "." + to_string(h4);
    
    return true;
}
```

#### 4.3 阅读 Client::downloadFile()

```cpp
// 文件位置：Client.cpp
// 函数：Client::downloadFile()
// 关键修复：先连接数据端口，再发送 RETR

bool Client::downloadFile(const string& filename) {
    // 1. 进入被动模式
    if(!pasvMode()) return false;
    
    // 2. 连接到数据端口（修复关键：在发送 RETR 之前！）
    dataSocket = socket(AF_INET, SOCK_STREAM, 0);
    connect(dataSocket, (struct sockaddr*)&dataAddr, ...);
    
    // 3. 发送 RETR 命令
    send(clientSocket, "RETR filename\r\n", ...);
    
    // 4. 读取 150 响应
    recv(clientSocket, buffer, ...);
    
    // 5. 接收文件数据
    while(true) {
        bytesReceived = recv(dataSocket, buffer, ...);
        if(bytesReceived <= 0) break;
        content.append(buffer, bytesReceived);
    }
    
    // 6. 保存文件
    ofstream file(filename, ios::binary);
    file.write(content.c_str(), content.length());
    
    return true;
}
```

```
  ├─→ opendir(newDir)
  │    └─→ 检查目录存在
  │
  ├─→ 更新 session->workingDir
  │
  └─→ 发送 250 响应
```

## 📚 进阶阅读

### 理解 epoll

```cpp
// epoll 三步骤

// 1. 创建 epoll 实例
int epollFd = epoll_create1(0);

// 2. 添加监听的套接字
struct epoll_event ev;
ev.events = EPOLLIN;  // 监听读事件
ev.data.fd = socketFd;
epoll_ctl(epollFd, EPOLL_CTL_ADD, socketFd, &ev);

// 3. 等待事件
struct epoll_event events[MAX_EVENTS];
int nfds = epoll_wait(epollFd, events, MAX_EVENTS, timeout);
```

### 理解 FTP 协议

**控制连接**：
- 端口 21（标准）或 2100（本实现）
- 传输 FTP 命令和响应
- 整个会话期间保持连接

**数据连接**：
- 动态端口（被动模式）
- 仅传输数据（文件内容、目录列表）
- 传输完成后关闭

**被动模式优势**：
- 客户端发起数据连接
- 适用于客户端在 NAT 后面的情况
- 防火墙配置更简单
